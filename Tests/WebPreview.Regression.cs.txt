    private static void Check(string name, bool pass)
    {
        _checks++; if (!pass) _failed++; Console.WriteLine($"{(pass ? "PASS" : "FAIL")} {name}");
    }
    private static void Pump()
    {
        var frame = new DispatcherFrame();
        Application.Current.Dispatcher.BeginInvoke(DispatcherPriority.ApplicationIdle, new Action(() => frame.Continue = false));
        Dispatcher.PushFrame(frame); foreach (Window window in Application.Current.Windows) window.UpdateLayout();
    }
    private static void Capture(Window window, string path)
    {
        window.UpdateLayout();
        var bitmap = new RenderTargetBitmap((int)Math.Ceiling(window.ActualWidth), (int)Math.Ceiling(window.ActualHeight), 96, 96, PixelFormats.Pbgra32);
        bitmap.Render(window);
        var encoder = new PngBitmapEncoder(); encoder.Frames.Add(BitmapFrame.Create(bitmap));
        using var stream = File.Create(path); encoder.Save(stream); Console.WriteLine("Screenshot: " + path);
    }

    private static void PumpFor(int milliseconds) => PumpUntil(() => false, milliseconds, false);
    private static void PumpUntil(Func<bool> predicate, int timeout = 4000, bool require = true)
    {
        var frame = new DispatcherFrame(); var start = DateTime.UtcNow;
        var timer = new DispatcherTimer { Interval = TimeSpan.FromMilliseconds(10) };
        timer.Tick += (_, _) => { if (predicate() || (DateTime.UtcNow - start).TotalMilliseconds >= timeout) frame.Continue = false; };
        timer.Start(); Dispatcher.PushFrame(frame); timer.Stop(); Pump();
        if (require && !predicate()) throw new TimeoutException("Owned UI preview did not reach its expected state.");
    }
    private sealed class FakeHandler(Func<HttpRequestMessage, CancellationToken, Task<HttpResponseMessage>> reply) : HttpMessageHandler
    {
        public int Count;
        public List<string> Urls = [];
        protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken token)
        {
            Interlocked.Increment(ref Count); lock (Urls) Urls.Add(request.RequestUri!.AbsoluteUri); return reply(request, token);
        }
    }
