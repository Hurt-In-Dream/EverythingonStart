using System.Runtime.InteropServices;
using System.Windows;
using System.Windows.Interop;
using System.Windows.Threading;

internal sealed class Probe
{
    private static readonly IntPtr HWND_TOPMOST = new(-1), HWND_NOTOPMOST = new(-2);
    private const int GWL_EXSTYLE = -20;
    private const long WS_EX_TOPMOST = 8;
    private const uint SWP_NOSIZE = 1, SWP_NOMOVE = 2, SWP_NOZORDER = 4, SWP_NOACTIVATE = 0x10, SWP_ASYNCWINDOWPOS = 0x4000;
    private readonly HashSet<IntPtr> _demotedFullscreenWindows = [];
    private Hook? _winKeyHookService = null;
    private readonly Shell _taskbarList = new();
    private static readonly HashSet<IntPtr> Owned = [];
    private static IntPtr _foreground;
    private static int _failed, _checks;
    private void EnsureTaskbarListInitialized() { }
    private static IntPtr GetForegroundWindow() => _foreground;
    private static int GetSystemMetrics(int index) => index == 0 ? 1280 : 720;
    private static bool IsShellDesktopOrCoreWindow(IntPtr hwnd) => false;
    private sealed class Hook { public IntPtr LastAppBeforeWinKeyHwnd = IntPtr.Zero; }
    private sealed class Shell { public void MarkFullscreenWindow(IntPtr hwnd, bool fullscreen) { Guard(hwnd); } }
    /* REAL_DEMOTE */
    /* REAL_RESTORE */

    [STAThread]
    private static int Main()
    {
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        Console.WriteLine("ISOLATION: exact project demote/restore bodies; real off-screen HWNDs and native Z-order; synthetic foreground/shell/monitor; no global keys, no user-window changes.");
        foreach (bool taskbarTopmost in new[] { true, false })
        foreach (bool originalTopmost in new[] { false, true })
        {
            var probe = new Probe();
            var full = Window(1280, 720, originalTopmost);
            var taskbar = Window(1280, 40, taskbarTopmost);
            var target = Window(320, 200, false);
            IntPtr fs = Hwnd(full), tb = Hwnd(taskbar), selected = Hwnd(target);
            Owned.UnionWith([fs, tb, selected]);
            _foreground = fs;
            string label = $"taskbarTop={taskbarTopmost}/originalTop={originalTopmost}";
            bool original = Topmost(fs);
            probe.DemoteFullscreenWindowsBelowTaskbar(IntPtr.Zero, tb, fs);
            Pump();
            bool afterDemote = Topmost(fs);
            bool taskbarAbove = Above(tb, fs);
            SetWindowPos(selected, IntPtr.Zero, 0, 0, 0, 0, SWP_NOMOVE | SWP_NOSIZE | SWP_NOACTIVATE);
            _foreground = selected; // Controlled Alt+Tab selection: don't activate any real desktop window.
            bool selectedAboveBeforeHide = Above(selected, fs);
            probe.RestoreDemotedFullscreenWindows();
            Pump();
            bool selectedAboveAfterHide = Above(selected, fs);
            Console.WriteLine($"OBSERVED {label}: originalTop={original}; demotedTop={afterDemote}; taskbarAbove={taskbarAbove}; selectedAboveBeforeHide={selectedAboveBeforeHide}; selectedAboveAfterHide={selectedAboveAfterHide}; restoredTop={Topmost(fs)}");
            Check($"{label}/demotion-not-topmost", !afterDemote);
            if (taskbarTopmost) Check($"{label}/taskbar-remains-above", taskbarAbove);
            Check($"{label}/selected-window-not-covered-before-hide", selectedAboveBeforeHide);
            Check($"{label}/selected-window-not-covered-after-hide", selectedAboveAfterHide);
            Check($"{label}/tracked-handles-cleared", probe._demotedFullscreenWindows.Count == 0);
            // With the setting disabled there is no demotion/restore work at all.
            var idle = new Probe();
            bool beforeIdle = Topmost(fs), beforeIdleAbove = Above(selected, fs);
            idle.RestoreDemotedFullscreenWindows();
            Check($"{label}/no-tracked-fullscreen-no-change", Topmost(fs) == beforeIdle && Above(selected, fs) == beforeIdleAbove && idle._demotedFullscreenWindows.Count == 0);
            full.Close(); taskbar.Close(); target.Close();
            Owned.Clear();
        }
        Console.WriteLine($"Summary: {_checks - _failed}/{_checks} passed");
        app.Shutdown();
        return _failed == 0 ? 0 : 1;
    }
    private static Window Window(double width, double height, bool topmost)
    {
        var window = new Window { Width = width, Height = height, Left = -20000, Top = -20000,
            WindowStyle = WindowStyle.None, ResizeMode = ResizeMode.NoResize, ShowActivated = false,
            ShowInTaskbar = false, Topmost = topmost, Content = new System.Windows.Controls.Border() };
        window.Show();
        return window;
    }
    private static IntPtr Hwnd(Window window) => new WindowInteropHelper(window).Handle;
    private static bool Topmost(IntPtr hwnd) => (GetWindowLongPtrW(hwnd, GWL_EXSTYLE).ToInt64() & WS_EX_TOPMOST) != 0;
    private static bool Above(IntPtr upper, IntPtr lower)
    {
        IntPtr current = lower;
        for (int i = 0; i < 2000 && current != IntPtr.Zero; i++)
        {
            current = GetWindow(current, 3);
            if (current == upper) return true;
        }
        return false;
    }
    private static void Check(string name, bool pass)
    {
        _checks++; if (!pass) _failed++;
        Console.WriteLine($"{(pass ? "PASS" : "FAIL")} {name}");
    }
    private static void Guard(IntPtr hwnd) { if (!Owned.Contains(hwnd)) throw new InvalidOperationException("Refusing to mutate a non-test HWND."); }
    private static bool SetWindowPos(IntPtr hwnd, IntPtr after, int x, int y, int w, int h, uint flags)
    {
        Guard(hwnd);
        bool result = NativeSetWindowPos(hwnd, after, x, y, w, h, flags);
        if (!result) throw new System.ComponentModel.Win32Exception(Marshal.GetLastWin32Error());
        return result;
    }
    private static void Pump()
    {
        var frame = new DispatcherFrame();
        Application.Current.Dispatcher.BeginInvoke(DispatcherPriority.ApplicationIdle, new Action(() => frame.Continue = false));
        Dispatcher.PushFrame(frame);
    }
    [StructLayout(LayoutKind.Sequential)] private struct RECT { public int Left, Top, Right, Bottom; }
    [DllImport("user32.dll", EntryPoint="SetWindowPos", SetLastError=true)] private static extern bool NativeSetWindowPos(IntPtr hwnd, IntPtr after, int x, int y, int w, int h, uint flags);
    [DllImport("user32.dll")] private static extern IntPtr GetWindowLongPtrW(IntPtr hwnd, int index);
    [DllImport("user32.dll")] private static extern bool GetWindowRect(IntPtr hwnd, out RECT rect);
    [DllImport("user32.dll")] private static extern bool IsWindowVisible(IntPtr hwnd);
    [DllImport("user32.dll")] private static extern bool IsWindow(IntPtr hwnd);
    [DllImport("user32.dll")] private static extern IntPtr GetWindow(IntPtr hwnd, uint command);
    [DllImport("user32.dll")] private static extern uint GetWindowThreadProcessId(IntPtr hwnd, out uint pid);
    [DllImport("kernel32.dll")] private static extern uint GetCurrentThreadId();
}
