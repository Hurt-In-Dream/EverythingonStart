using System.IO;
using System.Collections;
using System.Linq.Expressions;
using System.Reflection;
using System.Runtime.CompilerServices;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Media.Imaging;
using System.Windows.Threading;
using EverythingonStart;
using EverythingonStart.Services;

internal static class Program
{
    private const BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic;
    private static int _checks, _failed;
    [STAThread]
    private static int Main(string[] args)
    {
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        var wallpaperType = typeof(MainWindow).Assembly.GetType("EverythingonStart.Services.DesktopWallpaperService");
        var backdropType = typeof(MainWindow).Assembly.GetType("EverythingonStart.Services.WindowBackdropService")!;
        if (wallpaperType == null)
        {
            Console.WriteLine("BASELINE: background-source=screen-capture; original-start-menu-can-be-included; wallpaper-layer=absent");
            Check("desktop-wallpaper-layer-provider", wallpaperType != null);
            Check("screen-capture-method-removed", backdropType.GetMethod("CaptureDesktopRegion", BindingFlags.Static | BindingFlags.NonPublic) == null);
            Check("wallpaper-reader-wired", backdropType.GetField("_wallpaperReader", Private) != null);
            Check("explicit-wallpaper-source", backdropType.GetProperty("SourceDescription") != null);
            Console.WriteLine($"Summary: {_checks - _failed}/{_checks} passed"); app.Shutdown(); return 1;
        }
        Check("desktop-wallpaper-layer-provider", wallpaperType != null);
        Check("screen-capture-method-removed", backdropType.GetMethod("CaptureDesktopRegion", BindingFlags.Static | BindingFlags.NonPublic) == null);
        Check("wallpaper-reader-wired", backdropType.GetField("_wallpaperReader", Private) != null);
        Check("explicit-wallpaper-source", backdropType.GetProperty("SourceDescription") != null);
        var assembly = typeof(MainWindow).Assembly;
        Type providerType = wallpaperType!;
        var layoutType = assembly.GetType("EverythingonStart.Services.DesktopWallpaperLayout")!;
        var monitorType = assembly.GetType("EverythingonStart.Services.DesktopWallpaperMonitor")!;
        var stateType = assembly.GetType("EverythingonStart.Services.DesktopWallpaperState")!;
        var imageType = assembly.GetType("EverythingonStart.Services.DesktopWallpaperImage")!;
        var controlType = assembly.GetType("EverythingonStart.Controls.GaussianBlurBackdrop")!;
        object Monitor(Rect bounds, string path) => Activator.CreateInstance(monitorType, bounds, path)!;
        object State(int layout, Color background, bool enabled, params object[] monitors)
        {
            var values = Array.CreateInstance(monitorType, monitors.Length);
            for (int i = 0; i < monitors.Length; i++) values.SetValue(monitors[i], i);
            return Activator.CreateInstance(stateType, Enum.ToObject(layoutType, layout), background, values, enabled)!;
        }
        var redGreen = Image(200, 100, Colors.Red, Colors.Lime);
        var blue = Image(200, 100, Colors.Blue, Colors.Blue);
        object? ImageReader(string path) => path == "red-green" ? Activator.CreateInstance(imageType, redGreen, new Size(200, 100))
            : path == "blue" ? Activator.CreateInstance(imageType, blue, new Size(200, 100)) : null;
        Delegate ImageDelegate(Func<string, object?> reader)
        {
            var argument = System.Linq.Expressions.Expression.Parameter(typeof(string));
            var body = System.Linq.Expressions.Expression.Convert(System.Linq.Expressions.Expression.Invoke(System.Linq.Expressions.Expression.Constant(reader), argument), imageType);
            return System.Linq.Expressions.Expression.Lambda(typeof(Func<,>).MakeGenericType(typeof(string), imageType), body, argument).Compile();
        }
        var imageReader = ImageDelegate(ImageReader);
        BitmapSource Render(Rect region, object state) => (BitmapSource)providerType.GetMethod("RenderRegion")!.Invoke(null, [region, state, imageReader])!;
        Rect Placement(Size natural, Rect monitor, int mode) => (Rect)providerType.GetMethod("GetPlacement")!.Invoke(null, [natural, monitor, Enum.ToObject(layoutType, mode)])!;
        var square = new Rect(0, 0, 100, 100);
        Check("layout-center", Placement(new Size(200, 100), square, 0) == new Rect(-50, 0, 200, 100));
        Check("layout-stretch", Placement(new Size(200, 100), square, 2) == square);
        Check("layout-fit", Placement(new Size(200, 100), square, 3) == new Rect(0, 25, 100, 50));
        Check("layout-fill", Placement(new Size(200, 100), square, 4) == new Rect(-50, 0, 200, 100));
        var state = State(2, Colors.Black, true, Monitor(new Rect(0, 0, 200, 100), "red-green"), Monitor(new Rect(200, 0, 200, 100), "blue"));
        Check("left-desktop-region", ColorAt(Render(new Rect(0, 0, 20, 20), state), 10, 10) == Colors.Red);
        Check("right-desktop-region", ColorAt(Render(new Rect(180, 0, 20, 20), state), 10, 10) == Colors.Lime);
        Check("different-wallpaper-per-monitor", ColorAt(Render(new Rect(200, 0, 20, 20), state), 10, 10) == Colors.Blue);
        var across = Render(new Rect(190, 0, 20, 20), state);
        Check("cross-monitor-crop", ColorAt(across, 4, 10) == Colors.Lime && ColorAt(across, 16, 10) == Colors.Blue);
        var negative = State(2, Colors.Black, true, Monitor(new Rect(-200, -100, 200, 100), "red-green"));
        Check("negative-monitor-coordinates", ColorAt(Render(new Rect(-200, -100, 20, 20), negative), 10, 10) == Colors.Red);
        var fitted = Render(square, State(3, Colors.Blue, true, Monitor(square, "red-green")));
        Check("fit-preserves-desktop-background-bars", ColorAt(fitted, 10, 10) == Colors.Blue && ColorAt(fitted, 10, 50) == Colors.Red);
        var spanning = State(5, Colors.Black, true, Monitor(new Rect(0, 0, 100, 100), "red-green"), Monitor(new Rect(100, 0, 100, 100), "red-green"));
        var span = Render(new Rect(0, 0, 200, 100), spanning);
        Check("span-layout-across-monitors", ColorAt(span, 20, 50) == Colors.Red && ColorAt(span, 180, 50) == Colors.Lime);
        var tile = Render(new Rect(0, 0, 240, 30), State(1, Colors.Black, true, Monitor(new Rect(0, 0, 240, 30), "red-green")));
        Check("tile-layout", ColorAt(tile, 20, 10) == Colors.Red && ColorAt(tile, 150, 10) == Colors.Lime && ColorAt(tile, 220, 10) == Colors.Red);
        var disabled = Render(new Rect(0, 0, 20, 20), State(4, Colors.Purple, false, Monitor(square, "red-green")));
        Check("disabled-desktop-wallpaper-is-solid-color", ColorAt(disabled, 10, 10) == Colors.Purple);
        var missing = Render(new Rect(0, 0, 20, 20), State(4, Colors.Purple, true, Monitor(square, "missing")));
        Check("missing-image-uses-desktop-color-not-screen", ColorAt(missing, 10, 10) == Colors.Purple);
        Check("output-is-frozen", across.IsFrozen);
        Check("oversized-region-bounded", providerType.GetMethod("RenderRegion")!.Invoke(null, [new Rect(0, 0, 8192, 8192), state, imageReader]) == null);

        var settings = (AppSettingsService)RuntimeHelpers.GetUninitializedObject(typeof(AppSettingsService));
        typeof(AppSettingsService).GetProperty("Current")!.SetValue(settings, new AppSettings());
        typeof(AppSettingsService).GetField("_settingsFilePath", Private)!.SetValue(settings, Path.Combine(args[0], "isolated-settings.json"));
        typeof(AppSettings).GetProperty("EnableGaussianBlur")!.SetValue(settings.Current, true);
        settings.Current.WindowOpacityPercent = 75;
        var background = (FrameworkElement)Activator.CreateInstance(controlType)!;
        var window = new Window { WindowStyle = WindowStyle.None, AllowsTransparency = true, Background = Brushes.Transparent,
            Width = 200, Height = 160, Left = -9000, Top = -9000, ShowActivated = false, ShowInTaskbar = false, Content = background };
        int reads = 0; Rect last = Rect.Empty;
        Func<Rect, BitmapSource?> wallpaperReader = region => { reads++; last = region; return redGreen; };
        var service = Activator.CreateInstance(backdropType, window, background, settings, wallpaperReader, (Func<bool>)(() => false))!;
        window.Show(); PumpFor(150);
        Check("wallpaper-layer-visible", reads >= 1 && controlType.GetProperty("Snapshot")!.GetValue(background) != null);
        Check("source-identifies-wallpaper-not-screen", ((string)backdropType.GetProperty("SourceDescription")!.GetValue(service)!).Contains("壁纸"));
        int beforeMove = reads;
        double oldX = last.X; window.Left += 80; PumpFor(250);
        Check("moving-window-reprojects-wallpaper", reads > beforeMove && Math.Abs(last.X - oldX - 80) <= 1);
        int beforeResize = reads;
        window.Width += 40; PumpFor(250);
        Check("resizing-window-reprojects-wallpaper", reads > beforeResize && last.Width >= 240);
        int settled = reads; PumpFor(250);
        Check("no-idle-continuous-sampling", reads == settled);
        window.Hide(); PumpFor(150);
        Check("hide-clears-wallpaper-buffer", controlType.GetProperty("Snapshot")!.GetValue(background) == null);
        window.Close();
        Check("dispose-stops-refresh-timer", (bool)backdropType.GetField("_disposed", Private)!.GetValue(service)!);
        if (args.Contains("--live-wallpaper"))
        {
            var live = Activator.CreateInstance(providerType, new object?[] { null, null })!;
            var liveState = providerType.GetMethod("ReadWindowsWallpaper")!.Invoke(null, null);
            Check("live-explorer-wallpaper-getter", liveState != null && ((IEnumerable)stateType.GetProperty("Monitors")!.GetValue(liveState)!).Cast<object>().Any());
            var firstMonitor = ((IEnumerable)stateType.GetProperty("Monitors")!.GetValue(liveState)!).Cast<object>().First();
            var bounds = (Rect)monitorType.GetProperty("Bounds")!.GetValue(firstMonitor)!;
            var bitmap = (BitmapSource?)providerType.GetMethod("ReadRegion")!.Invoke(live, [new Rect(bounds.Left, bounds.Top, 8, 8)]);
            Console.WriteLine($"LIVE WALLPAPER RESULT: bitmap={bitmap != null}; diagnostic={providerType.GetField("_lastReadFailure", Private)!.GetValue(live)}");
            Check("live-wallpaper-render-memory-only", bitmap is { PixelWidth: 8, PixelHeight: 8, IsFrozen: true });
            if (bitmap != null) Console.WriteLine("LIVE WALLPAPER: Windows read-only wallpaper configuration and 8x8 in-memory projection verified; no paths/pixels/images printed or saved.");
            ((IDisposable)live).Dispose();
        }
        Console.WriteLine($"Summary: {_checks - _failed}/{_checks} passed"); app.Shutdown(); return _failed == 0 ? 0 : 1;
    }
    private static void Check(string name, bool pass)
    {
        _checks++; if (!pass) _failed++; Console.WriteLine($"{(pass ? "PASS" : "FAIL")} {name}");
    }
    private static BitmapSource Image(int width, int height, Color left, Color right)
    {
        byte[] pixels = new byte[width * height * 4];
        for (int y = 0; y < height; y++) for (int x = 0; x < width; x++)
        {
            Color color = x < width / 2 ? left : right; int i = (y * width + x) * 4;
            pixels[i] = color.B; pixels[i + 1] = color.G; pixels[i + 2] = color.R; pixels[i + 3] = 255;
        }
        var image = BitmapSource.Create(width, height, 96, 96, PixelFormats.Bgra32, null, pixels, width * 4); image.Freeze(); return image;
    }
    private static Color ColorAt(BitmapSource image, int x, int y)
    {
        byte[] pixel = new byte[4]; image.CopyPixels(new Int32Rect(x, y, 1, 1), pixel, 4, 0); return Color.FromArgb(pixel[3], pixel[2], pixel[1], pixel[0]);
    }
    private static void PumpFor(int milliseconds)
    {
        var frame = new DispatcherFrame(); var timer = new DispatcherTimer { Interval = TimeSpan.FromMilliseconds(milliseconds) };
        timer.Tick += (_, _) => frame.Continue = false; timer.Start(); Dispatcher.PushFrame(frame); timer.Stop();
    }
}
